import pygame import random import sys pygame.init() # Configurações largura = 400 altura = 600 tela = pygame.display.set_mode((largura, altura)) pygame.display.set_caption("Corrida de Carro - Grok") clock = pygame.time.Clock() fonte = pygame.font.SysFont("Arial", 28) # Cores BRANCO = (255, 255, 255) VERMELHO = (255, 0, 0) PRETO = (0, 0, 0) CINZA = (40, 40, 40) AMARELO = (255, 215, 0) # Carro do jogador carro_largura = 50 carro_altura = 80 carro_x = largura // 2 - carro_largura // 2 carro_y = altura - 120 velocidade_carro = 7 # Variáveis do jogo velocidade_estrada = 8 pontos = 0 game_over = False rodando = True obstaculos = [] moedas = [] def desenhar_estrada(): tela.fill(CINZA) pygame.draw.rect(tela, (30, 30, 30), (80, 0, largura-160, altura)) # pista # Faixas da estrada for i in range(-2, 12): y = (pygame.time.get_ticks() // 8 % 80) + i * 80 pygame.draw.rect(tela, BRANCO, (largura//2 - 5, y, 10, 40)) def desenhar_carro(x, y): pygame.draw.rect(tela, VERMELHO, (x, y, carro_largura, carro_altura)) pygame.draw.rect(tela, (0, 200, 255), (x+8, y+15, carro_largura-16, 30)) # vidro def criar_obstaculo(): x = random.randint(100, largura-100) obstaculos.append([x, -60, 50, 60]) def criar_moeda(): x = random.randint(100, largura-100) moedas.append([x, -30]) # Loop principal while rodando: for evento in pygame.event.get(): if evento.type == pygame.QUIT: rodando = False if evento.type == pygame.KEYDOWN and game_over: if evento.key == pygame.K_r: # Reiniciar obstaculos.clear() moedas.clear() pontos = 0 velocidade_estrada = 8 game_over = False if not game_over: teclas = pygame.key.get_pressed() if teclas[pygame.K_LEFT] and carro_x > 85: carro_x -= velocidade_carro if teclas[pygame.K_RIGHT] and carro_x < largura - carro_largura - 85: carro_x += velocidade_carro # Spawn if random.randint(1, 35) == 1: criar_obstaculo() if random.randint(1, 28) == 1: criar_moeda() # Atualizar obstáculos for obs in obstaculos[:]: obs[1] += velocidade_estrada if obs[1] > altura: obstaculos.remove(obs) pontos += 5 # Atualizar moedas for m in moedas[:]: m[1] += velocidade_estrada if m[1] > altura: moedas.remove(m) # Colisões for obs in obstaculos: if (carro_x < obs[0] + obs[2] and carro_x + carro_largura > obs[0] and carro_y < obs[1] + obs[3] and carro_y + carro_altura > obs[1]): game_over = True for i, m in enumerate(moedas): if abs(carro_x + carro_largura//2 - m[0]) < 35 and abs(carro_y + carro_altura//2 - m[1]) < 35: moedas.pop(i) pontos += 20 break if pontos > 0 and pontos % 300 == 0 and velocidade_estrada < 16: velocidade_estrada += 0.3 # Desenhar tudo desenhar_estrada() for obs in obstaculos: pygame.draw.rect(tela, (200, 50, 0), (obs[0], obs[1], obs[2], obs[3])) for m in moedas: pygame.draw.circle(tela, AMARELO, (m[0], m[1]), 14) pygame.draw.circle(tela, (200, 170, 0), (m[0], m[1]), 8) desenhar_carro(carro_x, carro_y) # HUD texto = fonte.render(f"Pontos: {pontos}", True, BRANCO) tela.blit(texto, (10, 10)) if game_over: go_texto = fonte.render("GAME OVER", True, (255, 50, 50)) tela.blit(go_texto, (largura//2 - go_texto.get_width()//2, altura//2 - 50)) restart = fonte.render("Pressione R para jogar novamente", True, BRANCO) tela.blit(restart, (largura//2 - restart.get_width()//2, altura//2 + 20)) pygame.display.flip() clock.tick(60) pygame.quit() sys.exit()
